cmake_minimum_required(VERSION 3.16)
project(wbr_ros2_bridge LANGUAGES CXX)
# UNBUILT on this Mac. Requires an already provisioned ROS 2 Jazzy environment.
message(WARNING "UNBUILT skeleton: snapshot ingress and clock mapper are not wired")
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(diagnostic_updater REQUIRED)
set(WBR_VIO_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../vio_adapter" CACHE PATH
    "Directory containing the actual EstimateSnapshot adapter.h")
if(NOT EXISTS "${WBR_VIO_INCLUDE_DIR}/adapter.h")
  message(FATAL_ERROR "Set WBR_VIO_INCLUDE_DIR to the Q1 adapter headers")
endif()
add_executable(snapshot_bridge src/bridge_node.cpp)
target_compile_features(snapshot_bridge PRIVATE cxx_std_17)
target_compile_options(snapshot_bridge PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(snapshot_bridge PRIVATE "${WBR_VIO_INCLUDE_DIR}")
ament_target_dependencies(snapshot_bridge rclcpp geometry_msgs diagnostic_msgs diagnostic_updater)
install(TARGETS snapshot_bridge DESTINATION lib/${PROJECT_NAME})
install(FILES DESIGN.md REPORT.md DESTINATION share/${PROJECT_NAME})
ament_package()
