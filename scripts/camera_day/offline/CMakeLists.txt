cmake_minimum_required(VERSION 3.16)
project(wbr_offline_stereo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenCV REQUIRED COMPONENTS core imgcodecs)
find_package(Eigen3 REQUIRED)
find_package(Pangolin REQUIRED)
find_package(Threads REQUIRED)
set(ORB_ROOT "" CACHE PATH "Isolated ORB-SLAM3 source/build root")
find_library(ORB_LIBRARY ORB_SLAM3 PATHS "${ORB_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
add_executable(replay_stereo replay_stereo.cc)
target_include_directories(replay_stereo PRIVATE "${ORB_ROOT}" "${ORB_ROOT}/include" "${ORB_ROOT}/include/CameraModels" "${ORB_ROOT}/Thirdparty/Sophus" ../native ${OpenCV_INCLUDE_DIRS} ${Pangolin_INCLUDE_DIRS})
target_link_libraries(replay_stereo PRIVATE "${ORB_LIBRARY}" Eigen3::Eigen ${OpenCV_LIBS} ${Pangolin_LIBRARIES} Threads::Threads)
