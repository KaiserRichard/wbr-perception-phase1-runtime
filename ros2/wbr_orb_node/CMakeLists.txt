cmake_minimum_required(VERSION 3.19)
project(wbr_orb_node LANGUAGES CXX)
option(WITH_ORB_SLAM3 "Use an existing native ORB-SLAM3 build (IMU and visual modes)" OFF)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(yaml_cpp_vendor REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(WBR_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(WBR_TOOLS "${WBR_REPO_ROOT}/tools")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS config/defaults.json scripts/generate_config.py)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/generate_config.py"
    "${CMAKE_CURRENT_SOURCE_DIR}/config/defaults.json" "${CMAKE_CURRENT_BINARY_DIR}/generated/defaults.hpp"
    COMMAND_ERROR_IS_FATAL ANY)
add_library(wbr_orb_core src/core.cpp "${WBR_TOOLS}/vio_adapter/adapter.cpp")
target_compile_features(wbr_orb_core PUBLIC cxx_std_17)
target_compile_options(wbr_orb_core PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(wbr_orb_core PUBLIC
    "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_CURRENT_BINARY_DIR}/generated"
    "${WBR_TOOLS}/vio_adapter" "${WBR_TOOLS}/ros2_bridge/src" "${WBR_TOOLS}/runtime_observer")
target_link_libraries(wbr_orb_core PUBLIC Threads::Threads)
add_executable(orb_node src/node.cpp)
ament_target_dependencies(orb_node rclcpp sensor_msgs nav_msgs yaml_cpp_vendor)
target_link_libraries(orb_node wbr_orb_core yaml-cpp)
if(WITH_ORB_SLAM3)
    set(ORB_SLAM3_ROOT "" CACHE PATH "Existing qualified native ORB checkout")
    set(ORB_EXTRA_INCLUDE_DIRS "" CACHE STRING "Additional Eigen/Sophus/Pangolin include directories")
    set(ORB_EXTRA_LIBRARIES "" CACHE STRING "Additional transitive native libraries")
    find_package(OpenCV REQUIRED COMPONENTS core)
    find_package(Eigen3 REQUIRED)
    find_library(ORB_SLAM3_LIBRARY ORB_SLAM3 PATHS "${ORB_SLAM3_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    if(NOT EXISTS "${ORB_SLAM3_ROOT}/include/System.h")
        message(FATAL_ERROR "ORB_SLAM3_ROOT must identify an existing qualified build")
    endif()
    target_sources(wbr_orb_core PRIVATE "${WBR_TOOLS}/vio_adapter/orb_engine.cpp")
    target_compile_definitions(wbr_orb_core PUBLIC WBR_WITH_ORBSLAM3)
    target_include_directories(wbr_orb_core PUBLIC "${ORB_SLAM3_ROOT}/include"
        "${ORB_SLAM3_ROOT}/include/CameraModels" "${ORB_SLAM3_ROOT}" ${OpenCV_INCLUDE_DIRS} ${ORB_EXTRA_INCLUDE_DIRS})
    target_link_libraries(wbr_orb_core PUBLIC "${ORB_SLAM3_LIBRARY}" Eigen3::Eigen ${OpenCV_LIBS} ${ORB_EXTRA_LIBRARIES})
    find_package(cv_bridge REQUIRED)
    find_package(geometry_msgs REQUIRED)
    find_package(std_msgs REQUIRED)
    add_executable(d435_node src/d435_node.cpp)
    target_compile_features(d435_node PRIVATE cxx_std_17)
    target_compile_options(d435_node PRIVATE -Wall -Wextra -Wpedantic)
    ament_target_dependencies(d435_node rclcpp sensor_msgs geometry_msgs std_msgs cv_bridge)
    target_link_libraries(d435_node wbr_orb_core)
    install(TARGETS d435_node DESTINATION lib/${PROJECT_NAME})
endif()
install(TARGETS orb_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md REPORT.md DESTINATION share/${PROJECT_NAME})
# Helpers run from this source worktree to reuse the existing tools without copying them.
ament_package()
